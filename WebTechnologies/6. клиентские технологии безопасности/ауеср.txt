fetch('http://server.com/flag', {
  method: 'PATCH',
  headers: {
    'MT-Priority': 'high'
  }
})
.then(response => {
  if (!response.ok) {
    throw new Error('Network response was not ok: ' + response.statusText);
  }
  return response.text();
})
.then(data => {
  console.log(data);
})
.catch(error => {
  console.log('Fetch error:', error);
});
